    puts("Введите трёхзначное число");

    first_num = n / 100;
    srednee_num = (n / 10) % 10;
    last_num = n % 10;
    sum_num = first_num + srednee_num + last_num;
    obratnoe_num = last_num * 100 + srednee_num * 10 + first_num;
